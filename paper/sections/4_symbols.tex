\section{符号说明}

表~\ref{tab:core-symbols}列出全文高频使用的核心符号，局部中间量在公式首次出现处定义。

\begin{table}[H]
  \centering
  \caption{核心符号说明}
  \label{tab:core-symbols}
  \small
  \begin{tabularx}{\textwidth}{>{\centering\arraybackslash}p{0.22\textwidth}>{\raggedright\arraybackslash}X>{\centering\arraybackslash}p{0.19\textwidth}>{\centering\arraybackslash}p{0.11\textwidth}}
    \toprule
    符号 & 含义 & 类型或取值域 & 适用问题 \\
    \midrule
    $p_0$ & 供应商声称的标称次品率，本题取 $0.10$ & $[0,1]$ & 一 \\
    $n,X$ & 固定抽样数与样本次品数 & 非负整数 & 一 \\
    $\gamma_R,\gamma_A$ & 拒收信度与接收信度 & $(0,1)$ & 一 \\
    $\alpha_R,\alpha_A$ & 拒收与接收检验的显著性水平 & $(0,1)$ & 一 \\
    $a_n,r_n$ & 接收上界与拒收下界 & 整数 & 一 \\
    $n_*$ & 首次同时存在互不重叠两类判定域的样本量 & 正整数 & 一 \\
    $D_0,D_1$ & 有限总体最大可接受与最小不可接受次品数 & 整数 & 一 \\
    $\omega_i$ & 在手零配件 $i$ 的真实质量状态 & $\{N,G,B\}$ & 二 \\
    $b(\omega_1,\omega_2)$ & 给定已有信息后的联合信念 & $[0,1]$ & 二 \\
    $\pi(b),V(b),Q(b,a)$ & 最优策略、状态价值与状态--动作价值 & 实数/元每件 & 二 \\
    $c_i,t_i,c_a,t_f,d$ & 采购、检测、装配、成品检测与拆解成本 & 非负实数/元每件 & 二、三 \\
    $S,L$ & 市场售价与不合格品调换损失 & 非负实数/元每件 & 二、三 \\
    $X$ & 16 位检测/拆解决策向量 $(x_1,\ldots,x_{16})$ & $\{0,1\}^{16}$ & 三 \\
    $s$ & 零配件或半成品的质量状态，取值 $\{N,KG,UG,UB\}$ & 四态集合 & 三 \\
    $V_X,C_X,P_X$ & 固定策略下状态 $S$ 的期望成本、一步成本与转移概率 & 实数/概率 & 三 \\
    $E[\Pi(X)]$ & 固定策略 $X$ 的期望利润 & 实数/元每件 & 三 \\
    $\hat p,k,n$ & 某环节样本次品率、样本次品数与抽样样本量 & $[0,1]$,整数 & 四 \\
    $\pi(p\mid k,n)$ & 次品率的 Beta 后验密度 & 密度函数 & 四 \\
    $\theta^{(s)},\Pi(X,\theta)$ & 第 $s$ 组后验情景与策略 $X$ 在该参数下的期望利润 & 向量,实数/元每件 & 四 \\
    $\bar\Pi,\sigma,q_{0.05}$ & 后验期望利润、利润标准差与 $5\%$ 分位数 & 实数/元每件 & 四 \\
    \bottomrule
  \end{tabularx}
\end{table}
